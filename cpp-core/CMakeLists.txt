cmake_minimum_required(VERSION 3.16)
project(FPMPortable VERSION 1.0.0 LANGUAGES C CXX)

option(FPM_BUILD_TESTS "Build portable core tests" ON)
option(FPM_WITH_OPENCV "Use OpenCV for CLI image codecs and comparison tests" OFF)
option(FPM_BUILD_GUI "Build the Qt6 desktop GUI" OFF)
option(FPM_DISABLE_SIMD "Compile only the portable scalar dot product" OFF)
option(FPM_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(FPM_BUILD_EXPERIMENTS "Build isolated float/RGB similarity and GPU experiments" OFF)

add_library(fpm_core STATIC src/image.cpp src/ncc.cpp src/geometry.cpp src/subpixel.cpp src/matcher.cpp src/candidate_dedup.cpp)
target_include_directories(fpm_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(fpm_core PUBLIC cxx_std_17)
set_target_properties(fpm_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(fpm_c SHARED src/c_api.cpp)
target_link_libraries(fpm_c PRIVATE fpm_core)
target_include_directories(fpm_c PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(fpm_c PRIVATE FPM_BUILDING_LIBRARY PUBLIC FPM_SHARED)
if(FPM_DISABLE_SIMD)
    target_compile_definitions(fpm_core PRIVATE FPM_DISABLE_SIMD)
endif()
# x86-64 only: the highest tier the core may pick at run time. VNNI_INT8 drops to
# AVX2 when the compiler cannot build AVX-VNNI-INT8 intrinsics.
set(FPM_X86_MAX_ISA "VNNI_INT8" CACHE STRING "Highest x86-64 SIMD tier selected at run time: SSE2, AVX2 or VNNI_INT8")
set_property(CACHE FPM_X86_MAX_ISA PROPERTY STRINGS SSE2 AVX2 VNNI_INT8)
if(FPM_X86_MAX_ISA STREQUAL "SSE2")
    set(fpm_x86_level 1)
elseif(FPM_X86_MAX_ISA STREQUAL "AVX2")
    set(fpm_x86_level 2)
elseif(FPM_X86_MAX_ISA STREQUAL "VNNI_INT8")
    include(CheckCXXSourceCompiles)
    check_cxx_source_compiles([=[
#include <immintrin.h>
#if defined(_MSC_VER) && !defined(__clang__)
#define FPM_TARGET
#else
#define FPM_TARGET __attribute__((target("avx2,avxvnniint8")))
#endif
FPM_TARGET static __m128i dot(__m128i a, __m128i b) { return _mm_dpbuud_epi32(a, a, b); }
int main() { return _mm_cvtsi128_si32(dot(_mm_setzero_si128(), _mm_setzero_si128())); }
]=] FPM_HAVE_AVX_VNNI_INT8)
    if(FPM_HAVE_AVX_VNNI_INT8)
        set(fpm_x86_level 3)
    else()
        set(fpm_x86_level 2)
    endif()
else()
    message(FATAL_ERROR "FPM_X86_MAX_ISA must be SSE2, AVX2 or VNNI_INT8")
endif()
target_compile_definitions(fpm_core PRIVATE FPM_X86_MAX_ISA=${fpm_x86_level})
foreach(target fpm_core fpm_c)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(FPM_SANITIZERS AND NOT MSVC)
    target_compile_options(fpm_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fpm_core PUBLIC -fsanitize=address,undefined)
endif()

add_library(fpm_cli_support STATIC tools/cli.cpp)
target_link_libraries(fpm_cli_support PUBLIC fpm_core)
target_include_directories(fpm_cli_support PUBLIC tools)
add_executable(fpm_cli tools/cli_main.cpp)
target_link_libraries(fpm_cli PRIVATE fpm_cli_support)
if(FPM_WITH_OPENCV)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
    target_compile_definitions(fpm_cli_support PRIVATE FPM_WITH_OPENCV)
    target_include_directories(fpm_cli_support PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(fpm_cli_support PRIVATE ${OpenCV_LIBS})
endif()
if(FPM_BUILD_GUI)
    find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent)
    add_executable(fpm_gui ../macos-cli/main.cpp ../macos-cli/mainwindow.cpp ../macos-cli/mainwindow.h)
    set_target_properties(fpm_gui PROPERTIES AUTOMOC ON)
    if(APPLE)
        set_target_properties(fpm_gui PROPERTIES MACOSX_BUNDLE ON
            MACOSX_BUNDLE_GUI_IDENTIFIER "org.fpm.portable"
            MACOSX_BUNDLE_BUNDLE_NAME "Image Pattern Matching")
    endif()
    target_link_libraries(fpm_gui PRIVATE fpm_cli_support Qt6::Widgets Qt6::Concurrent)
endif()
if(FPM_BUILD_TESTS)
    enable_testing()
    add_executable(fpm_core_test tests/core_test.cpp)
    target_include_directories(fpm_core_test PRIVATE src)
    target_link_libraries(fpm_core_test PRIVATE fpm_core fpm_c)
    add_test(NAME portable_core COMMAND fpm_core_test)
    add_executable(fpm_candidate_dedup_test tests/candidate_dedup_test.cpp)
    target_include_directories(fpm_candidate_dedup_test PRIVATE src)
    target_link_libraries(fpm_candidate_dedup_test PRIVATE fpm_core)
    add_test(NAME research_candidate_dedup COMMAND fpm_candidate_dedup_test)
    add_executable(fpm_c_api_test tests/c_api_test.c)
    target_link_libraries(fpm_c_api_test PRIVATE fpm_c)
    add_test(NAME c_abi COMMAND fpm_c_api_test)
    if(FPM_WITH_OPENCV)
        add_executable(fpm_opencv_test tests/opencv_test.cpp)
        target_include_directories(fpm_opencv_test PRIVATE src ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(fpm_opencv_test PRIVATE fpm_core ${OpenCV_LIBS})
        add_test(NAME opencv_stage_comparison COMMAND fpm_opencv_test)
        # Frozen local macOS port, retained only as an audit reference.
        add_executable(fpm_legacy_reference tests/legacy_reference.cpp ../macos-cli/match_core.cpp)
        target_include_directories(fpm_legacy_reference PRIVATE ../macos-cli ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(fpm_legacy_reference PRIVATE ${OpenCV_LIBS})
        if(APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
            target_compile_options(fpm_legacy_reference PRIVATE -march=armv8.4-a+dotprod)
        endif()
    endif()
    if(FPM_BUILD_GUI)
        add_executable(fpm_qt_gray_loader_test tests/qt_gray_loader_test.cpp)
        target_include_directories(fpm_qt_gray_loader_test PRIVATE tools)
        target_link_libraries(fpm_qt_gray_loader_test PRIVATE fpm_core Qt6::Gui)
        add_test(NAME qt_gray_loader COMMAND fpm_qt_gray_loader_test)
        find_package(Qt6 REQUIRED COMPONENTS Test)
        add_executable(fpm_gui_test tests/gui_test.cpp ../macos-cli/mainwindow.cpp ../macos-cli/mainwindow.h)
        set_target_properties(fpm_gui_test PROPERTIES AUTOMOC ON)
        target_include_directories(fpm_gui_test PRIVATE ../macos-cli)
        target_compile_definitions(fpm_gui_test PRIVATE FPM_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../Test Images")
        target_link_libraries(fpm_gui_test PRIVATE fpm_core Qt6::Widgets Qt6::Concurrent Qt6::Test)
        add_test(NAME gui_workflow COMMAND fpm_gui_test)
        set_tests_properties(gui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()

if(FPM_BUILD_EXPERIMENTS)
    add_subdirectory(experiments)
    add_executable(fpm_ncc_pruning_benchmark tools/benchmark_ncc_pruning.cpp)
    target_include_directories(fpm_ncc_pruning_benchmark PRIVATE src)
    target_link_libraries(fpm_ncc_pruning_benchmark PRIVATE fpm_core)
endif()

include(GNUInstallDirs)
install(TARGETS fpm_core fpm_c fpm_cli EXPORT FPMTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES ../LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/fpm)
