find_package(Threads REQUIRED)
add_library(fpm_pose_experiment SHARED pose_ncc.cpp)
target_compile_features(fpm_pose_experiment PUBLIC cxx_std_17)
target_link_libraries(fpm_pose_experiment PRIVATE Threads::Threads)
target_include_directories(fpm_pose_experiment PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(fpm_pose_experiment PRIVATE FPM_EXPERIMENT_BUILD)
if(APPLE)
    enable_language(OBJCXX)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/pose_ncc.metal")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/pose_ncc.metal" FPM_METAL_SOURCE)
    configure_file(metal_source.hpp.in metal_source.hpp @ONLY)
    target_sources(fpm_pose_experiment PRIVATE pose_metal.mm)
    target_include_directories(fpm_pose_experiment PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
    target_compile_definitions(fpm_pose_experiment PRIVATE FPM_EXPERIMENT_METAL)
    set_source_files_properties(pose_metal.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(fpm_pose_experiment PRIVATE "-framework Metal" "-framework Foundation")
endif()
if(FPM_BUILD_TESTS)
    add_executable(fpm_pose_test ../tests/pose_experiment_test.cpp)
    target_link_libraries(fpm_pose_test PRIVATE fpm_pose_experiment)
    add_test(NAME experimental_pose_cpu COMMAND fpm_pose_test)
    # Portable test factory simulates unavailable hardware versus real shader
    # failures. Compile the solver directly; do not link a real Metal backend.
    add_executable(fpm_pose_auto_fallback_test pose_ncc.cpp ../tests/pose_auto_fallback_test.cpp)
    target_compile_features(fpm_pose_auto_fallback_test PRIVATE cxx_std_17)
    target_include_directories(fpm_pose_auto_fallback_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(fpm_pose_auto_fallback_test PRIVATE FPM_EXPERIMENT_BUILD FPM_EXPERIMENT_METAL)
    target_link_libraries(fpm_pose_auto_fallback_test PRIVATE Threads::Threads)
    add_test(NAME experimental_pose_auto_fallback COMMAND fpm_pose_auto_fallback_test)
    if(APPLE)
        add_executable(fpm_pose_linearization_test ../tests/pose_linearization_test.cpp)
        target_link_libraries(fpm_pose_linearization_test PRIVATE fpm_pose_experiment)
        target_compile_definitions(fpm_pose_linearization_test PRIVATE FPM_EXPERIMENT_METAL)
        add_test(NAME experimental_pose_metal_derivatives COMMAND fpm_pose_linearization_test)
        set_tests_properties(experimental_pose_metal_derivatives PROPERTIES SKIP_RETURN_CODE 77)
    endif()
endif()
if(MSVC)
    target_compile_options(fpm_pose_experiment PRIVATE /W4 /utf-8)
else()
    target_compile_options(fpm_pose_experiment PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FPM_SANITIZERS AND NOT MSVC)
    # Consumers must use the same libc++ container annotations as this library:
    # mixed sanitized/unsanitized weak vector instantiations can leave valid
    # elements poisoned. Propagate instrumentation rather than disable checks.
    target_compile_options(fpm_pose_experiment PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fpm_pose_experiment PUBLIC -fsanitize=address,undefined)
    if(TARGET fpm_pose_auto_fallback_test)
        target_compile_options(fpm_pose_auto_fallback_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(fpm_pose_auto_fallback_test PRIVATE -fsanitize=address,undefined)
    endif()
endif()
