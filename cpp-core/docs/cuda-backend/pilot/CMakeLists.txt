cmake_minimum_required(VERSION 3.24)
project(fpm_cuda_pilot LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 120)
set(FPM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${FPM_SOURCE} fpm)
add_executable(cuda_pilot cuda_pilot.cu ref.cpp)
target_include_directories(cuda_pilot PRIVATE ${FPM_SOURCE}/src)
target_link_libraries(cuda_pilot PRIVATE fpm_core)
target_compile_options(cuda_pilot PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-fmad=false -lineinfo -Xcompiler=/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
